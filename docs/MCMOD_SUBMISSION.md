```powershell
python scripts/prepare-mcmod-package.py `
  --items run-neoforge/let_me_see_see/echo_warrior.json `
  --release-jar build/release-0.2.2-published/artifacts/neoforge/build/libs/echo-warrior-neoforge-26.1.2-0.2.2.jar `
```
